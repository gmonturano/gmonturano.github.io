SAFE JOINT WALD TEST # ============================================================================== joint_wald_test <- function( model, V, terms) { if ( inherits(model, "error") || is.null(V) ) { return( tibble::tibble( wald_stat = NA_real_, wald_df = NA_integer_, wald_p = NA_real_ ) ) } b <- stats::coef( model ) terms <- intersect( terms, names(b) ) if (length(terms) == 0L) { return( tibble::tibble( wald_stat = NA_real_, wald_df = 0L, wald_p = NA_real_ ) ) } bb <- b[ terms ] VV <- V[ terms, terms, drop = FALSE ] invV <- tryCatch( solve(VV), error = function(e) NULL ) if (is.null(invV)) { return( tibble::tibble( wald_stat = NA_real_, wald_df = length(terms), wald_p = NA_real_ ) ) } W <- as.numeric( t(bb) %*% invV %*% bb ) df <- length( terms ) p <- stats::pchisq( W, df = df, lower.tail = FALSE ) tibble::tibble( wald_stat = W, wald_df = df, wald_p = p ) } # ============================================================================== # 19. FIT ONE TWFE MODEL # ============================================================================== fit_one_nonlinear <- function( P, formula, test_terms, dataset_name, y, x, spec, model_name, pipeline_name) { empty_result <- function(message, model_object = NULL, V = NULL) { n_obs <- if (!is.null(model_object)) { tryCatch( as.integer(stats::nobs(model_object)), error = function(e) nrow(P) ) } else { nrow(P) } list( terms = tibble::tibble( dataset = dataset_name, y = y, x = x, spec = spec, model = model_name, pipeline = pipeline_name, term = NA_character_, beta = NA_real_, se = NA_real_, stat = NA_real_, p = NA_real_, ci_low = NA_real_, ci_high = NA_real_, n = n_obs, error = message ), joint = tibble::tibble( dataset = dataset_name, y = y, x = x, spec = spec, model = model_name, pipeline = pipeline_name, wald_stat = NA_real_, wald_df = NA_integer_, wald_p = NA_real_, error = message ), model_object = model_object, vcov = V ) } m <- tryCatch( plm::plm( formula, data = P, model = "within", effect = "twoways" ), error = function(e) e ) if (inherits(m, "error")) { return( empty_result( m$message ) ) } V <- tryCatch( plm::vcovHC( m, method = "arellano", type = "HC1", cluster = "group" ), error = function(e) NULL ) if (is.null(V)) { return( empty_result( "vcovHC cluster-group failed", model_object = m ) ) } b <- tryCatch( stats::coef(m), error = function(e) NULL ) if (is.null(b) || length(b) == 0L) { return( empty_result( "No coefficients returned by plm.", model_object = m, V = V ) ) } # Do not depend on coeftest() column names or dimensions. # Align coefficients and covariance matrix explicitly by term name. if ( is.null(rownames(V)) || is.null(colnames(V)) ) { return( empty_result( "Robust covariance matrix has no coefficient names.", model_object = m, V = V ) ) } common_terms <- Reduce( intersect, list( names(b), rownames(V), colnames(V) ) ) if (length(common_terms) == 0L) { return( empty_result( "No coefficient names matched the covariance matrix.", model_object = m, V = V ) ) } b_use <- b[ common_terms ] V_use <- V[ common_terms, common_terms, drop = FALSE ] variances <- diag( V_use ) se <- sqrt( ifelse( is.finite(variances) & variances >= 0, variances, NA_real_ ) ) stat <- b_use / se # Same asymptotic two-sided normal inference used for robust coefficient # reporting; no assumption about the printed structure of coeftest(). p <- 2 * stats::pnorm( abs(stat), lower.tail = FALSE ) terms_out <- tibble::tibble( dataset = dataset_name, y = y, x = x, spec = spec, model = model_name, pipeline = pipeline_name, term = common_terms, beta = as.numeric(b_use), se = as.numeric(se), stat = as.numeric(stat), p = as.numeric(p), ci_low = as.numeric(b_use) - 1.96 * as.numeric(se), ci_high = as.numeric(b_use) + 1.96 * as.numeric(se), n = as.integer( stats::nobs(m) ), error = NA_character_ ) JT <- joint_wald_test( model = m, V = V, terms = test_terms ) joint_out <- tibble::tibble( dataset = dataset_name, y = y, x = x, spec = spec, model = model_name, pipeline = pipeline_name, wald_stat = JT$wald_stat, wald_df = JT$wald_df, wald_p = JT$wald_p, error = NA_character_ ) list( terms = terms_out, joint = joint_out, model_object = m, vcov = V ) } # BUILD EXPOSURE-COMPONENT CURVE # ============================================================================== build_curve <- function( model_object, model_name, pipeline_name, dataset_name, y, x, spec, frozen_data, x_original) { if (is.null(model_object)) { return( tibble::tibble() ) } b <- stats::coef( model_object ) xv <- as_num( frozen_data[[x_original]] ) xv <- xv[ is.finite(xv) ] if (length(xv) < 2L) { return( tibble::tibble() ) } q <- stats::quantile( xv, probs = c(0.01, 0.99), na.rm = TRUE, names = FALSE ) if ( !all(is.finite(q)) || q[1] >= q[2] ) { q <- range( xv, na.rm = TRUE ) } x_grid <- seq( q[1], q[2], length.out = 100L ) xbar <- attr( frozen_data, "x_mean" ) xc <- x_grid - xbar component <- rep( 0, length(x_grid) ) if (identical(model_name, "linear")) { if ("X_CENTERED" %in% names(b)) { component <- component + b[["X_CENTERED"]] * xc } } else if (identical(model_name, "quadratic")) { if ("X_CENTERED" %in% names(b)) { component <- component + b[["X_CENTERED"]] * xc } if ("X_SQ" %in% names(b)) { component <- component + b[["X_SQ"]] * xc^2 } } else if (identical(model_name, "cubic")) { if ("X_CENTERED" %in% names(b)) { component <- component + b[["X_CENTERED"]] * xc } if ("X_SQ" %in% names(b)) { component <- component + b[["X_SQ"]] * xc^2 } if ("X_CUBE" %in% names(b)) { component <- component + b[["X_CUBE"]] * xc^3 } } else if ( identical(model_name, "spline_df3") && isTRUE(attr(frozen_data, "ns3_ok")) ) { knots <- attr( frozen_data, "ns3_knots" ) boundary <- attr( frozen_data, "ns3_boundary" ) B <- tryCatch( splines::ns( xc, knots = knots, Boundary.knots = boundary, intercept = FALSE ), error = function(e) NULL ) if (is.null(B)) { return(tibble::tibble()) } B <- as.matrix(B) for (j in seq_len(ncol(B))) { nm <- paste0( "NS3_", j ) if (nm %in% names(b)) { component <- component + b[[nm]] * B[, j] } } } else if ( identical(model_name, "spline_df4") && isTRUE(attr(frozen_data, "ns4_ok")) ) { knots <- attr( frozen_data, "ns4_knots" ) boundary <- attr( frozen_data, "ns4_boundary" ) B <- tryCatch( splines::ns( xc, knots = knots, Boundary.knots = boundary, intercept = FALSE ), error = function(e) NULL ) if (is.null(B)) { return(tibble::tibble()) } B <- as.matrix(B) for (j in seq_len(ncol(B))) { nm <- paste0( "NS4_", j ) if (nm %in% names(b)) { component <- component + b[[nm]] * B[, j] } } } # Normalize the displayed component to zero at the first grid point. component <- component - component[1] tibble::tibble( dataset = dataset_name, y = y, x = x, spec = spec, model = model_name, pipeline = pipeline_name, exposure = x_grid, fitted_component = component ) } # ============================================================================== # 21. CHOOSE SPECIFICATIONS # ============================================================================== get_specs_to_run <- function( spec_blocks, dataset_name) { nm <- names( spec_blocks ) if (is.null(nm) || length(nm) == 0L) { stop( paste0( dataset_name, ": spec_blocks has no named specifications." ), call. = FALSE ) } baseline <- if ("S0_FE_ONLY" %in% nm) { "S0_FE_ONLY" } else { warning( paste0( dataset_name, ": S0_FE_ONLY not found. Using first specification: ", nm[1] ) ) nm[1] } full_spec <- tail( nm, 1 ) unique( c( baseline, full_spec ) ) } # ============================================================================== # 23. RUN EVERYTHING # ============================================================================== ALL_TERMS <- list() ALL_JOINT <- list() ALL_DIAG <- list() ALL_CURVES <- list() ALL_ERRORS <- list() counter <- 0L for (dataset_name in names(DESIGN)) { DD <- DESIGN[[dataset_name]] specs_to_run <- get_specs_to_run( DD$spec_blocks, dataset_name ) cat( "\n============================================================\n", "DATASET: ", dataset_name, "\n", "SPECS: ", paste( specs_to_run, collapse = " | " ), "\n", "============================================================\n", sep = "" ) for (y_name in DD$outcomes) { for (x_name in DD$exposures) { for (spec_name in specs_to_run) { counter <- counter + 1L cat( "\nCELL ", counter, " | ", dataset_name, " | Y=", y_name, " | X=", x_name, " | SPEC=", spec_name, "\n", sep = "" ) OBJ <- tryCatch( run_one_cell( dataset_name = dataset_name, DDESIGN = DD, y = y_name, x = x_name, spec = spec_name ), error = function(e) { ALL_ERRORS[[length(ALL_ERRORS) + 1L]] <<- tibble::tibble( dataset = dataset_name, y = y_name, x = x_name, spec = spec_name, error = e$message ) cat( "CELL ERROR | ", e$message, "\n", sep = "" ) NULL } ) if (is.null(OBJ)) { next } ALL_TERMS[[length(ALL_TERMS) + 1L]] <- OBJ$terms ALL_JOINT[[length(ALL_JOINT) + 1L]] <- OBJ$joint ALL_DIAG[[length(ALL_DIAG) + 1L]] <- OBJ$diagnostics ALL_CURVES[[length(ALL_CURVES) + 1L]] <- OBJ$curves } } } } TERMS <- dplyr::bind_rows( ALL_TERMS ) JOINT <- dplyr::bind_rows( ALL_JOINT ) DIAGNOSTICS <- dplyr::bind_rows( ALL_DIAG ) CURVES <- dplyr::bind_rows( ALL_CURVES ) ERRORS <- dplyr::bind_rows( ALL_ERRORS ) # ============================================================================== # 24. SAVE ERRORS IMMEDIATELY # ============================================================================== write.csv( ERRORS, file.path( OUT_DIR, "04_NONLINEAR_ERRORS.csv" ), row.names = FALSE ) # ============================================================================== # 25. FAIL FAST IF NOTHING WAS ESTIMATED # ============================================================================== if (nrow(TERMS) == 0L) { cat( "\n============================================================\n", "NONLINEAR RUN FAILED BEFORE ESTIMATION OUTPUT\n", "============================================================\n\n", sep = "" ) if (nrow(ERRORS) > 0L) { print( ERRORS, n = Inf, width = Inf ) } else { cat( "No explicit cell error was recorded.\n" ) } stop( paste0( "TERMS is empty. Inspect: ", file.path( OUT_DIR, "04_NONLINEAR_ERRORS.csv" ) ), call. = FALSE ) } # ============================================================================== # 26. SAVE RAW RESULTS # ============================================================================== write.csv( TERMS, file.path( OUT_DIR, "01_NONLINEAR_ALL_TERMS.csv" ), row.names = FALSE ) write.csv( JOINT, file.path( OUT_DIR, "02_NONLINEAR_JOINT_TESTS.csv" ), row.names = FALSE ) write.csv( DIAGNOSTICS, file.path( OUT_DIR, "03_NONLINEAR_INDEX_DIAGNOSTICS.csv" ), row.names = FALSE ) write.csv( CURVES, file.path( OUT_DIR, "04B_NONLINEAR_PLOT_DATA.csv" ), row.names = FALSE ) # ============================================================================== # 27. TERM-BY-TERM ROUTE COMPARISON # # Base column extraction is used here deliberately to avoid tidyselect # external-vector ambiguity. # ============================================================================== required_term_cols <- c( "dataset", "y", "x", "spec", "model", "term", "pipeline", "beta", "se", "p", "n", "error" ) missing_term_cols <- setdiff( required_term_cols, names(TERMS) ) if (length(missing_term_cols) > 0L) { stop( paste0( "TERMS missing columns: ", paste( missing_term_cols, collapse = ", " ) ), call. = FALSE ) } ROUTE_BASE <- TERMS[ , required_term_cols, drop = FALSE ] ROUTE_COMPARE <- ROUTE_BASE %>% tidyr::pivot_wider( names_from = pipeline, values_from = c( beta, se, p, n, error ) ) %>% dplyr::mutate( delta_beta_FILTER_SUBSET = beta_FILTER - beta_SUBSET, delta_beta_REBUILT_SUBSET = beta_REBUILT - beta_SUBSET, abs_delta_FILTER_SUBSET = abs( delta_beta_FILTER_SUBSET ), abs_delta_REBUILT_SUBSET = abs( delta_beta_REBUILT_SUBSET ), sign_flip_FILTER_SUBSET = dplyr::case_when( is.finite(beta_FILTER) & is.finite(beta_SUBSET) ~ sign(beta_FILTER) != sign(beta_SUBSET), TRUE ~ NA ), sig_FILTER = dplyr::case_when( is.finite(p_FILTER) ~ p_FILTER < 0.05, TRUE ~ NA ), sig_SUBSET = dplyr::case_when( is.finite(p_SUBSET) ~ p_SUBSET < 0.05, TRUE ~ NA ), significance_flip_FILTER_SUBSET = dplyr::case_when( !is.na(sig_FILTER) & !is.na(sig_SUBSET) ~ sig_FILTER != sig_SUBSET, TRUE ~ NA ) ) write.csv( ROUTE_COMPARE, file.path( OUT_DIR, "05_NONLINEAR_ROUTE_COMPARISON.csv" ), row.names = FALSE ) # ============================================================================== # 28. JOINT-TEST ROUTE COMPARISON # ============================================================================== required_joint_cols <- c( "dataset", "y", "x", "spec", "model", "pipeline", "wald_stat", "wald_df", "wald_p", "error" ) missing_joint_cols <- setdiff( required_joint_cols, names(JOINT) ) if (length(missing_joint_cols) > 0L) { stop( paste0( "JOINT missing columns: ", paste( missing_joint_cols, collapse = ", " ) ), call. = FALSE ) } JOINT_BASE <- JOINT[ , required_joint_cols, drop = FALSE ] JOINT_COMPARE <- JOINT_BASE %>% tidyr::pivot_wider( names_from = pipeline, values_from = c( wald_stat, wald_df, wald_p, error ) ) %>% dplyr::mutate( sig_FILTER = ifelse( is.finite( wald_p_FILTER ), wald_p_FILTER < 0.05, NA ), sig_SUBSET = ifelse( is.finite( wald_p_SUBSET ), wald_p_SUBSET < 0.05, NA ), joint_significance_flip = ifelse( !is.na(sig_FILTER) & !is.na(sig_SUBSET), sig_FILTER != sig_SUBSET, NA ) ) write.csv( JOINT_COMPARE, file.path( OUT_DIR, "06_NONLINEAR_JOINT_ROUTE_COMPARISON.csv" ), row.names = FALSE ) # ============================================================================== # 29. COMPACT SUMMARY # ============================================================================== SUMMARY_TERMS <- ROUTE_COMPARE %>% dplyr::group_by( dataset, model, spec ) %>% dplyr::summarise( estimable_terms = sum( is.finite(beta_FILTER) & is.finite(beta_SUBSET) ), nonzero_filter_subset = sum( abs_delta_FILTER_SUBSET > 1e-10, na.rm = TRUE ), sign_flips = sum( sign_flip_FILTER_SUBSET %in% TRUE, na.rm = TRUE ), significance_flips = sum( significance_flip_FILTER_SUBSET %in% TRUE, na.rm = TRUE ), max_abs_filter_subset = ifelse( any( is.finite( abs_delta_FILTER_SUBSET ) ), max( abs_delta_FILTER_SUBSET, na.rm = TRUE ), NA_real_ ), max_abs_rebuilt_subset = ifelse( any( is.finite( abs_delta_REBUILT_SUBSET ) ), max( abs_delta_REBUILT_SUBSET, na.rm = TRUE ), NA_real_ ), .groups = "drop" ) write.csv( SUMMARY_TERMS, file.path( OUT_DIR, "07_NONLINEAR_SUMMARY.csv" ), row.names = FALSE ) # ============================================================================== # 30. RECOVERY VALIDATION # ============================================================================== RECOVERY_SOURCE <- ROUTE_COMPARE[ is.finite( ROUTE_COMPARE$delta_beta_REBUILT_SUBSET ), , drop = FALSE ] if (nrow(RECOVERY_SOURCE) > 0L) { RECOVERY_VALIDATION <- tibble::tibble( estimable_comparisons = nrow( RECOVERY_SOURCE ), max_abs_rebuilt_subset = max( abs( RECOVERY_SOURCE$delta_beta_REBUILT_SUBSET ), na.rm = TRUE ), rebuilt_equals_subset = max( abs( RECOVERY_SOURCE$delta_beta_REBUILT_SUBSET ), na.rm = TRUE ) < 1e-10 ) } else { RECOVERY_VALIDATION <- tibble::tibble( estimable_comparisons = 0L, max_abs_rebuilt_subset = NA_real_, rebuilt_equals_subset = FALSE ) } write.csv( RECOVERY_VALIDATION, file.path( OUT_DIR, "08_NONLINEAR_RECOVERY_VALIDATION.csv" ), row.names = FALSE ) # ============================================================================== # 31. INDEX DIAGNOSTIC SUMMARY # ============================================================================== INDEX_SUMMARY <- DIAGNOSTICS %>% dplyr::distinct( dataset, y, x, spec, pipeline, .keep_all = TRUE ) %>% dplyr::arrange( dataset, y, x, spec, pipeline ) write.csv( INDEX_SUMMARY, file.path( OUT_DIR, "09_NONLINEAR_INDEX_SUMMARY.csv" ), row.names = FALSE ) # ============================================================================== # 32. CURVE PLOTS # ============================================================================== if (nrow(CURVES) > 0L) { model_labels <- c( linear = "Linear", quadratic = "Quadratic", cubic = "Cubic", spline_df3 = "Natural spline (df = 3)", spline_df4 = "Natural spline (df = 4)" ) CURVES$model_label <- unname( model_labels[ CURVES$model ] ) CURVES$model_label <- factor( CURVES$model_label, levels = unname( model_labels ) ) plot_keys <- unique( CURVES[ , c( "dataset", "y", "x", "spec" ), drop = FALSE ] ) for (i in seq_len(nrow(plot_keys))) { key <- plot_keys[ i, , drop = FALSE ] Z <- CURVES[ CURVES$dataset == key$dataset & CURVES$y == key$y & CURVES$x == key$x & CURVES$spec == key$spec, , drop = FALSE ] if (nrow(Z) == 0L) { next } p <- ggplot2::ggplot( Z, ggplot2::aes( x = exposure, y = fitted_component, linetype = pipeline ) ) + ggplot2::geom_line( linewidth = 0.8 ) + ggplot2::facet_wrap( ~ model_label, scales = "free_y", ncol = 2 ) + ggplot2::labs( title = NULL, subtitle = paste( key$dataset, "|", key$y, "|", key$x, "|", key$spec ), x = key$x, y = "Estimated exposure component", linetype = "Pipeline" ) + ggplot2::theme_minimal( base_size = 10 ) + ggplot2::theme( legend.position = "bottom" ) stem <- safe_name( paste( "NONLINEAR_CURVES", key$dataset, key$y, key$x, key$spec, sep = "__" ) ) ggplot2::ggsave( filename = file.path( FIG_DIR, paste0( stem, ".png" ) ), plot = p, width = 10, height = 8, dpi = 300 ) ggplot2::ggsave( filename = file.path( FIG_DIR, paste0( stem, ".pdf" ) ), plot = p, width = 10, height = 8 ) } } # ============================================================================== # 33. GLOBAL DIAGNOSTIC PLOT: FILTER-SUBSET MAXIMUM DIFFERENCE # ============================================================================== if (nrow(SUMMARY_TERMS) > 0L) { P_DELTA <- SUMMARY_TERMS[ is.finite( SUMMARY_TERMS$max_abs_filter_subset ), , drop = FALSE ] if (nrow(P_DELTA) > 0L) { p_delta <- ggplot2::ggplot( P_DELTA, ggplot2::aes( x = model, y = max_abs_filter_subset, shape = dataset ) ) + ggplot2::geom_point( size = 2.5, position = ggplot2::position_dodge( width = 0.35 ) ) + ggplot2::facet_wrap( ~ spec, scales = "free_y" ) + ggplot2::labs( title = NULL, x = NULL, y = "Maximum |FILTER - SUBSET| coefficient difference", shape = "Dataset" ) + ggplot2::theme_minimal( base_size = 10 ) + ggplot2::theme( axis.text.x = ggplot2::element_text( angle = 30, hjust = 1 ), legend.position = "bottom" ) ggplot2::ggsave( filename = file.path( FIG_DIR, "NONLINEAR_SUMMARY_MAX_DELTA.png" ), plot = p_delta, width = 9, height = 6, dpi = 300 ) ggplot2::ggsave( filename = file.path( FIG_DIR, "NONLINEAR_SUMMARY_MAX_DELTA.pdf" ), plot = p_delta, width = 9, height = 6 ) } } # ============================================================================== # 34.
